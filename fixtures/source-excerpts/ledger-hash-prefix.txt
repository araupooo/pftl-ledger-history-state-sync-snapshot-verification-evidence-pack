constexpr std::uint32_t
make_hash_prefix(char a, char b, char c)
{
    return (static_cast<std::uint32_t>(a) << 24) +
        (static_cast<std::uint32_t>(b) << 16) +
        (static_cast<std::uint32_t>(c) << 8);
}
